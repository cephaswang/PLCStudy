位置控制模式（PT Mode）— ST 程式與 HMI 模擬

左側為 Structured Text (ST) 控制邏輯範例，右側為對應的 HMI 操作面板模擬。可切換相對／絕對定位，觀察伺服致能、警報重置與脈波定位的互動關係。

SIMULATION · 前端模擬，非真實連線 PLC

PLC 程式（Structured Text）

PRG_ServoPositionControl.st 
(* ============================================================
   位置控制模式 (PT Mode) - Structured Text 範例程式
   對應伺服驅動器 + PLC 脈波輸出架構
   接線: PLC Y0 -> 驅動器 PULSE 端子 ; PLC Y1 -> 驅動器 SIGN 端子
   參數: P1-00 = 脈波+方向 / 無濾波 / 正邏輯 / 開集極接線
         P1-44 = 電子齒輪比分子   P1-45 = 電子齒輪比分母
   ============================================================ *)

PROGRAM PRG_ServoPositionControl
VAR
    (* --- HMI 控制指令 --- *)
    bSON_CMD      : BOOL;              (* 伺服致能 *)
    bARST_CMD     : BOOL;              (* 警報重置 *)
    bStart_CMD    : BOOL;              (* 啟動移動 *)
    bStop_CMD     : BOOL;              (* 停止 *)
    bAbsMode      : BOOL;              (* FALSE=相對移動  TRUE=絕對移動 *)

    (* --- 移動參數 --- *)
    dwTargetPulse : DINT   := 0;       (* 目標脈波數／目標位置 *)
    dwFrequency   : DWORD  := 5000;    (* 脈波頻率 Hz *)
    dwAccelTime   : DWORD  := 200;     (* 加減速時間 ms *)

    (* --- 狀態回授 --- *)
    bServoReady   : BOOL;              (* 對應 SRDY *)
    bAlarm        : BOOL;              (* 對應 ALRM *)
    bBusy         : BOOL;              (* 移動中旗標 *)
    lCurrentPos   : DINT   := 0;       (* 目前位置（累計脈波） *)

    (* --- 輸出／內部使用 --- *)
    Y_SON         : BOOL;
    Y_ARST        : BOOL;
    Y_PULSE       : BOOL;
    Y_SIGN        : BOOL;
    bMoveTrig     : BOOL;
    M_PositionDone: BOOL;
END_VAR

(* 1. 伺服致能：無警報時才允許送出 SON *)
Y_SON := bSON_CMD AND NOT bAlarm;

(* 2. 警報重置 *)
IF bARST_CMD THEN
    Y_ARST := TRUE;
    bAlarm := FALSE;
ELSE
    Y_ARST := FALSE;
END_IF;

(* 3. 停止指令：立即停止脈波輸出 *)
IF bStop_CMD THEN
    bBusy     := FALSE;
    bMoveTrig := FALSE;
END_IF;

(* 4. 移動指令觸發：伺服就緒、無警報、且未在動作中 *)
IF bStart_CMD AND bServoReady AND NOT bAlarm AND NOT bBusy THEN
    bMoveTrig := TRUE;
END_IF;

IF bMoveTrig THEN
    bBusy := TRUE;
    IF bAbsMode THEN
        (* 絕對定位，相當於梯形圖 DDRVA 指令 *)
        DDRVA(dwFrequency, dwTargetPulse, dwAccelTime, Y_PULSE, Y_SIGN);
    ELSE
        (* 相對定位，相當於梯形圖 DDRVI 指令 *)
        DDRVI(dwFrequency, dwTargetPulse, dwAccelTime, Y_PULSE, Y_SIGN);
    END_IF;
    bMoveTrig := FALSE;
END_IF;

(* 5. 定位完成後清除 Busy 旗標
      實際應用中改以驅動器 TPOS 訊號或軸控制器完成旗標判斷 *)
IF bBusy AND M_PositionDone THEN
    bBusy := FALSE;
END_IF;

END_PROGRAM
