(*================================================================
實作題2：紅綠燈交通號誌控制（簡化版）－ ST（結構化文字）版
機型：三菱 FX5CPU / FX5U　開發軟體：GX Works3
對應來源：0909RGY.csv（V1.2 修正版梯形圖，邏輯完全對應轉換）
專案名稱：0909RGY_ST
================================================================*)

(*
VAR
X0          : BOOL;    // 啟動開關（維持型，X0=ON運轉／OFF停止）
Y0          : BOOL;    // 綠燈輸出
Y1          : BOOL;    // 黃燈輸出
Y2          : BOOL;    // 紅燈輸出

M0          : BOOL;    // 綠燈階段旗標
M1          : BOOL;    // 黃燈階段旗標
M2          : BOOL;    // 紅燈階段旗標

TMR0        : TON;     // 綠燈計時 5.0秒
TMR1        : TON;     // 黃燈計時 2.0秒
TMR2        : TON;     // 紅燈計時 5.0秒

X0_TRIG     : R_TRIG;  // X0上升緣偵測（取代梯形圖之 ORP X0，僅供開機首次啟動用）
END_VAR
*)


(*----------------------------------------------------------------
以下程式敘述順序完全比照 0909RGY.csv 梯形圖的掃描順序撰寫，
確保每一行運算式所讀取到的 M0/M1/M2/TMR0.Q/TMR1.Q/TMR2.Q 是「前次值」
或「本次已更新值」，皆與原梯形圖的LD/OR/ANI/AND/OUT掃描結果一致。
----------------------------------------------------------------*)

(* 0. 偵測 X0 上升緣，對應：ORP X0 *)
X0_TRIG(CLK := X0);

(* 1. 綠燈階段旗標 M0
對應：LD T2 / OR M0 / ORP X0 / ANI M1 / AND X0 / OUT M0
條件：(T2完成 OR M0自保 OR X0上升緣) AND 非黃燈 AND 運轉中 *)
M0 := (TMR2.Q OR M0 OR X0_TRIG.Q) AND NOT M1 AND X0;

(* 2. 黃燈階段旗標 M1
對應：LD T0 / OR M1 / ANI M2 / AND X0 / OUT M1
條件：(T0完成 OR M1自保) AND 非紅燈 AND 運轉中
註：此處 M2 為「前次掃描值」，與原梯形圖此rung位於M2更新之前一致 *)
M1 := (TMR0.Q OR M1) AND NOT M2 AND X0;

(* 3. 紅燈階段旗標 M2
對應：LD T1 / OR M2 / ANI M0 / AND X0 / OUT M2
條件：(T1完成 OR M2自保) AND 非綠燈 AND 運轉中
註：此處 M0 為「本次已更新值」，與原梯形圖此rung位於M0更新之後一致 *)
M2 := (TMR1.Q OR M2) AND NOT M0 AND X0;

(* 4. 燈號輸出，對應：LD M0/OUT Y0、LD M1/OUT Y1、LD M2/OUT Y2 *)
Y0 := M0;
Y1 := M1;
Y2 := M2;

(* 5. 計時器，對應：OUT T0 K50 / OUT T1 K20 / OUT T2 K50
K50=5.0秒、K20=2.0秒（100ms基準），此處直接以T#5S/T#2S表示實際時間 *)
TMR0(IN := M0, PT := T#5S);
TMR1(IN := M1, PT := T#2S);
TMR2(IN := M2, PT := T#5S);

(*----------------------------------------------------------------
動作說明（與原文件邏輯一致）：
- X0 OFF → M0/M1/M2 全部因 AND X0 條件不成立而變為0，
Y0/Y1/Y2 全部熄滅，T0/T1/T2 的 IN 變為FALSE，ET自動歸零。
- X0 由 OFF→ON：X0_TRIG.Q 產生一次TRUE脈波，驅動M0導通，
Y0亮、T0開始計時，之後綠→黃→紅→綠 無限循環。
- 若需高精度計時，可將PT改為更精確之時間常數，
或於PLC參數中調整計時器解析度。
================================================================*)
